function(fviz_add_example target source)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE FEAViz::Core)
    # Some examples call C math functions directly. FEAViz links libm privately,
    # so Unix example executables must state their own dependency.
    if(NOT WIN32)
        target_link_libraries(${target} PRIVATE m)
    endif()
    fviz_apply_compiler_options(${target})
    fviz_apply_sanitizers(${target})
    fviz_set_target_output_directories(${target})
endfunction()

function(fviz_add_fea_example target source)
    if(NOT TARGET FEAVizFEA)
        return()
    endif()
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE FEAViz::FEA)
    if(NOT WIN32)
        target_link_libraries(${target} PRIVATE m)
    endif()
    fviz_apply_compiler_options(${target})
    fviz_apply_sanitizers(${target})
    fviz_set_target_output_directories(${target})
endfunction()
