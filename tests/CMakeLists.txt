function(fviz_add_test target source test_name)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE FEAViz::Core)
    fviz_apply_compiler_options(${target})
    fviz_apply_sanitizers(${target})
    add_test(NAME ${test_name} COMMAND ${target})
endfunction()

function(fviz_add_fea_test target source test_name)
    if(NOT TARGET FEAVizFEA)
        return()
    endif()
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE FEAViz::FEA)
    fviz_apply_compiler_options(${target})
    fviz_apply_sanitizers(${target})
    add_test(NAME ${test_name} COMMAND ${target})
endfunction()

# Compile public headers in isolation to keep the installed C API self-contained.
file(GLOB_RECURSE _fviz_public_headers RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../include" "${CMAKE_CURRENT_SOURCE_DIR}/../include/FViz/*.h")
list(APPEND _fviz_public_headers "FViz/FVizConfig.h" "FViz/FVizVersion.h")
enable_language(CXX)
foreach(_fviz_header IN LISTS _fviz_public_headers)
    string(MAKE_C_IDENTIFIER "${_fviz_header}" _fviz_header_id)
    set(_fviz_header_source "${CMAKE_CURRENT_BINARY_DIR}/header_${_fviz_header_id}.c")
    file(WRITE "${_fviz_header_source}" "#include <${_fviz_header}>\nint main(void) { return 0; }\n")
    add_library("FVizHeader_${_fviz_header_id}" OBJECT "${_fviz_header_source}")
    set(_fviz_header_link_target FEAViz::Core)
    if(TARGET FEAVizFEA AND (_fviz_header MATCHES "^FViz/FEA/FViz(FEA|IntegrationPointData|ResultField|ResultDatabase|PrimaryVariable|DeformedShape)"))
        set(_fviz_header_link_target FEAViz::FEA)
    endif()
    target_link_libraries("FVizHeader_${_fviz_header_id}" PRIVATE ${_fviz_header_link_target})
    fviz_apply_compiler_options("FVizHeader_${_fviz_header_id}")
    set(_fviz_header_cpp_source "${CMAKE_CURRENT_BINARY_DIR}/header_${_fviz_header_id}.cpp")
    file(WRITE "${_fviz_header_cpp_source}" "#include <${_fviz_header}>\nint main() { return 0; }\n")
    add_library("FVizHeaderCpp_${_fviz_header_id}" OBJECT "${_fviz_header_cpp_source}")
    target_link_libraries("FVizHeaderCpp_${_fviz_header_id}" PRIVATE ${_fviz_header_link_target})
    target_compile_features("FVizHeaderCpp_${_fviz_header_id}" PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options("FVizHeaderCpp_${_fviz_header_id}" PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options("FVizHeaderCpp_${_fviz_header_id}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
